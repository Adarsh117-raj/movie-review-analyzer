def clean_text(text):
    text = str(text).lower()
    text = re.sub(r"http\\S+|www\\S+", " ", text)
    text = re.sub(r"[^a-zA-Z0-9\\s']", " ", text)
    text = re.sub(r"\\s+", " ", text).strip()
    return text

  • Compound score >= 0.05: Positive
  • Compound score <= -0.05: Negative
  • Otherwise: Neutral

def analyze_sentiment(text):
    scores = sia.polarity_scores(str(text))
    compound = scores["compound"]

    if compound >= 0.05:
        sentiment = "Positive"
    elif compound <= -0.05:
        sentiment = "Negative"
    else:
        sentiment = "Neutral"

    return pd.Series({
        "positive_score": scores["pos"],
        "neutral_score": scores["neu"],
        "negative_score": scores["neg"],
        "compound_score": compound,
        "predicted_sentiment": sentiment
    })

sentiment_results = reviews_df["review"].apply(analyze_sentiment)
results_df = pd.concat([reviews_df, sentiment_results], axis=1)

def analyze_multiple_reviews(text):

    reviews = [
        line.strip()
        for line in text.split("\n")
        if line.strip()
    ]

    if not reviews:
        return pd.DataFrame(), None

    data = []

    for review in reviews:

        scores = sia.polarity_scores(review)

        compound = scores["compound"]

        if compound >= 0.05:
            sentiment = "Positive"
        elif compound <= -0.05:
            sentiment = "Negative"
        else:
            sentiment = "Neutral"

        data.append({
            "Movie Review": review,
            "Sentiment": sentiment,
            "Compound Score": compound
        })

    df = pd.DataFrame(data)

    counts = df["Sentiment"].value_counts().reindex(
        ["Positive", "Neutral", "Negative"],
        fill_value=0
    )

    fig = px.pie(
        names=counts.index,
        values=counts.values,
        title="Movie Review Sentiment Distribution",
        hole=0.4,
        color=counts.index,
        color_discrete_map={
            "Positive": "#16A34A",
            "Neutral": "#F59E0B",
            "Negative": "#DC2626"
        }
    )
